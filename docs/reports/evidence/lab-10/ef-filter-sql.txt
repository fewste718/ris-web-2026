SELECT s.id, s.number, s.title, s.equipment_id, e.name, c.id, c.name, s.description, s.status::text, s.created_by_user_id, s.assignee_user_id, u.full_name, s.created_at FROM support_tickets AS s INNER JOIN equipment AS e ON s.equipment_id = e.id INNER JOIN clients AS c ON e.client_id = c.id LEFT JOIN users AS u ON s.assignee_user_id = u.id WHERE s.status = @__filter_0 ORDER BY s.id
